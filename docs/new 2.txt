# 1. Start Triton container mounting the local model repository
docker run -d --name ccvnn_triton \
  --shm-size=2g \
  -p 8000:8000 -p 8001:8001 -p 8002:8002 \
  -v /home/runner/ccvnn-workspace/model:/models \
  nvcr.io/nvidia/tritonserver:26.08-py3 \
  tritonserver --model-repository=/models

# 2. Check startup logs to confirm both models loaded successfully
docker logs ccvnn_triton -f --tail 50
