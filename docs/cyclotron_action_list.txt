.github/workflows/windows-rdp.yml:

name: Azure Windows 11/Server RDP Environment

on:
  workflow_dispatch:
    inputs:
      session_hours:
        description: 'Duration to keep session active (in hours, max 6)'
        required: false
        default: '6'

jobs:
  rdp-session:
    runs-on: windows-latest
    timeout-minutes: 360

    steps:
      - name: Enable Remote Desktop & Configure Firewall
        shell: powershell
        run: |
          # Enable Remote Desktop in Registry
          Set-ItemProperty -Path 'HKLM:\System\CurrentControlSet\Control\Terminal Server' -Name "fDenyTSConnections" -Value 0
          
          # Enable Firewall Rules for RDP
          Enable-NetFirewallRule -DisplayGroup "Remote Desktop"
          
          # Require NLA Authentication
          Set-ItemProperty -Path 'HKLM:\System\CurrentControlSet\Control\Terminal Server\WinStations\RDP-Tcp' -Name "UserAuthentication" -Value 1

      - name: Set Admin Account Password
        shell: powershell
        run: |
          $Pass = "${{ secrets.RDP_PASSWORD }}"
          if (-not $Pass) {
            # Fallback default password if secret is omitted
            $Pass = "AzureRDP2026!SecurePass"
          }
          $SecurePassword = ConvertTo-SecureString $Pass -AsPlainText -Force
          $User = Get-LocalUser -Name "runneradmin"
          $User | Set-LocalUser -Password $SecurePassword

      - name: Install & Configure Ngrok Tunnel
        shell: powershell
        run: |
          # Download official Ngrok binary
          Invoke-WebRequest -Uri "https://bin.equinox.io/c/b3511111111/ngrok-v3-stable-windows-amd64.zip" -OutFile "ngrok.zip"
          Expand-Archive -Path "ngrok.zip" -DestinationPath "." -Force
          
          # Apply Ngrok Authtoken
          if (-not "${{ secrets.NGROK_AUTH_TOKEN }}") {
            Write-Error "NGROK_AUTH_TOKEN secret is missing in repository settings."
            exit 1
          }
          .\ngrok.exe authtoken ${{ secrets.NGROK_AUTH_TOKEN }}

      - name: Launch Tunnel & Print Connection Endpoint
        shell: powershell
        run: |
          # Start Ngrok TCP tunnel on default RDP port 3389
          Start-Process -FilePath ".\ngrok.exe" -ArgumentList "tcp 3389" -NoNewWindow
          Start-Sleep -Seconds 5
          
          # Query Ngrok local API to extract host and assigned port
          $TunnelInfo = Invoke-RestMethod -Uri "http://127.0.0.1:4040/api/tunnels"
          $RdpEndpoint = $TunnelInfo.tunnels[0].public_url -replace 'tcp://', ''
          
          Write-Host ""
          Write-Host "=========================================================" -ForegroundColor Green
          Write-Host "   WINDOWS AZURE RDP SESSION IS ACTIVE"                    -ForegroundColor Green
          Write-Host "=========================================================" -ForegroundColor Green
          Write-Host " Connect Address : $RdpEndpoint"                           -ForegroundColor Yellow
          Write-Host " Username        : runneradmin"                           -ForegroundColor Yellow
          Write-Host " Password        : [Configured via RDP_PASSWORD Secret]"   -ForegroundColor Cyan
          Write-Host "=========================================================" -ForegroundColor Green
          Write-Host ""

      - name: Maintain Active Session Loop
        shell: powershell
        run: |
          $Hours = [int]"${{ github.event.inputs.session_hours }}"
          $TotalSeconds = $Hours * 3600
          $Elapsed = 0
          
          Write-Host "Session running for $Hours hour(s). Do not cancel the workflow."
          while ($Elapsed -lt $TotalSeconds) {
            Start-Sleep -Seconds 60
            $Elapsed += 60
          }

          # Enable Firewall Rules for RDP
          Enable-NetFirewallRule -DisplayGroup "Remote Desktop"
          
          # Require NLA Authentication
          Set-ItemProperty -Path 'HKLM:\System\CurrentControlSet\Control\Terminal Server\WinStations\RDP-Tcp' -Name "UserAuthentication" -Value 1

      - name: Set Admin Account Password
        shell: powershell
        run: |
          $Pass = "${{ secrets.RDP_PASSWORD }}"
          if (-not $Pass) {
            # Fallback default password if secret is omitted
            $Pass = "AzureRDP2026!SecurePass"
          }
          $SecurePassword = ConvertTo-SecureString $Pass -AsPlainText -Force
          $User = Get-LocalUser -Name "runneradmin"
          $User | Set-LocalUser -Password $SecurePassword

      - name: Download & Install Tailscale
        shell: powershell
        run: |
          # Download official Tailscale MSI installer
          $InstallerPath = "$env:TEMP\tailscale.msi"
          Invoke-WebRequest -Uri "https://pkgs.tailscale.com/stable/tailscale-setup-latest.exe" -OutFile "tailscale-setup.exe"
          
          # Silent installation
          Start-Process -FilePath ".\tailscale-setup.exe" -ArgumentList "/quiet /norestart" -Wait
          
          # Add Tailscale binary path to current session env
          $env:Path += ";C:\Program Files\Tailscale"
          [Environment]::SetEnvironmentVariable("Path", $env:Path, [EnvironmentVariableTarget]::Machine)

      - name: Connect to Tailscale Tailnet
        shell: powershell
        run: |
          $env:Path += ";C:\Program Files\Tailscale"
          
          if (-not "${{ secrets.TAILSCALE_AUTHKEY }}") {
            Write-Error "TAILSCALE_AUTHKEY secret is missing in repository settings."
            exit 1
          }

          # Authenticate node using reusable/ephemeral auth key
          & "C:\Program Files\Tailscale\tailscale.exe" up --authkey=${{ secrets.TAILSCALE_AUTHKEY }} --hostname=github-runner-rdp --accept-routes

      - name: Display RDP Connection Information
        shell: powershell
        run: |
          $env:Path += ";C:\Program Files\Tailscale"
          
          # Retrieve IPv4 assigned to this node on Tailscale network
          $TailscaleIP = (& "C:\Program Files\Tailscale\tailscale.exe" ip -4).Trim()
          
          Write-Host ""
          Write-Host "=========================================================" -ForegroundColor Green
          Write-Host "   TAILSCALE MESH VPN RDP SESSION IS ACTIVE"             -ForegroundColor Green
          Write-Host "=========================================================" -ForegroundColor Green
          Write-Host " Connect Address : $TailscaleIP:3389"                     -ForegroundColor Yellow
          Write-Host " Hostname        : github-runner-rdp"                      -ForegroundColor Yellow
          Write-Host " Username        : runneradmin"                           -ForegroundColor Yellow
          Write-Host " Password        : [Configured via RDP_PASSWORD Secret]"   -ForegroundColor Cyan
          Write-Host "=========================================================" -ForegroundColor Green
          Write-Host ""

      - name: Maintain Active Session Loop
        shell: powershell
        run: |
          $Hours = [int]"${{ github.event.inputs.session_hours }}"
          $TotalSeconds = $Hours * 3600
          $Elapsed = 0
          
          Write-Host "Session running for $Hours hour(s). Do not cancel the workflow."
          while ($Elapsed -lt $TotalSeconds) {
            Start-Sleep -Seconds 60
            $Elapsed += 60
          }

        description: 'Subnet routes to advertise (e.g. 10.0.0.0/24, 192.168.1.0/24)'
        required: false
        default: ''

jobs:
  rdp-session:
    runs-on: windows-latest
    timeout-minutes: 360

    steps:
      - name: Enable Remote Desktop & OpenSSH
        shell: powershell
        run: |
          # Enable Remote Desktop in Registry
          Set-ItemProperty -Path 'HKLM:\System\CurrentControlSet\Control\Terminal Server' -Name "fDenyTSConnections" -Value 0
          
          # Enable Firewall Rules for RDP and OpenSSH
          Enable-NetFirewallRule -DisplayGroup "Remote Desktop"
          
          # Require NLA Authentication
          Set-ItemProperty -Path 'HKLM:\System\CurrentControlSet\Control\Terminal Server\WinStations\RDP-Tcp' -Name "UserAuthentication" -Value 1

          # Ensure Windows Capability for OpenSSH Server is active (required for Tailscale SSH)
          Add-WindowsCapability -Online -Name OpenSSH.Server~~~~0.0.1.0 -ErrorAction SilentlyContinue
          Start-Service sshd -ErrorAction SilentlyContinue
          Set-Service -Name sshd -StartupType 'Automatic' -ErrorAction SilentlyContinue

      - name: Set Admin Account Password
        shell: powershell
        run: |
          $Pass = "${{ secrets.RDP_PASSWORD }}"
          if (-not $Pass) {
            $Pass = "AzureRDP2026!SecurePass"

import numpy as np
import tritonclient.grpc as grpcclient

def run_ccvnn_inference(server_url="localhost:8001", batch_size=4):
    """
    Sends an inference request to Triton Server over gRPC for the CCVNN model.
    """
    try:
        triton_client = grpcclient.InferenceServerClient(url=server_url)
    except Exception as e:
        print(f"Failed to create Triton gRPC client: {e}")
        return

    # Match configuration defined in config.pbtxt
    model_name = "ccvnn"
    input_name = "input_real_imag"
    output_name = "output_logits"

    # Shape: [batch_size, 2, 64, 64] -> FP32 (Channel 0: Real, Channel 1: Imaginary)
    input_data = np.random.randn(batch_size, 2, 64, 64).astype(np.float32)

    # Prepare input tensor object
    inputs = [grpcclient.InferInput(input_name, input_data.shape, "FP32")]
    inputs[0].set_data_from_numpy(input_data)

    # Prepare requested output object
    outputs = [grpcclient.InferRequestedOutput(output_name)]

    print(f"Submitting batch of {batch_size} to Triton gRPC endpoint ({server_url})...")
    
    # Execute synchronous inference call
    results = triton_client.infer(
        model_name=model_name,
        inputs=inputs,
        outputs=outputs
    )

    # Parse response array
    output_logits = results.as_numpy(output_name)
    
    print("\n--- Inference Response Received ---")
    print(f"Output Tensor Shape : {output_logits.shape}")
    print(f"Sample Logits Output :\n{output_logits}")

if __name__ == "__main__":
    # Pass Tailscale IP (e.g., '100.x.y.z:8001') when querying runner remotely over mesh VPN
    run_ccvnn_inference(server_url="localhost:8001", batch_size=4)

class CCVNNV17Backbone(nn.Module):
    def __init__(self, input_dim=17, hidden_dim=64, output_dim=1):
        super(CCVNNV17Backbone, self).__init__()
        self.net = nn.Sequential(
            nn.Linear(input_dim, hidden_dim),
            nn.Hardswish(),  # Or custom ACON activation module
            nn.Linear(hidden_dim, hidden_dim // 2),
            nn.Hardswish(),
            nn.Linear(hidden_dim // 2, output_dim),
            nn.Sigmoid()     # Produces hazard probability [0.0, 1.0]
        )

    def forward(self, v_input_17):
        return self.net(v_input_17)
